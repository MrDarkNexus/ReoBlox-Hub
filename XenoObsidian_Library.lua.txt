local function ParentGuiSafe(gui)
	local ok, hui = pcall(function()
		return (gethui and gethui()) or nil
	end)
	if ok and hui then
		gui.Parent = hui
		return
	end
	local okCore = pcall(function()
		gui.Parent = CoreGui
	end)
	if not okCore then
		gui.Parent = LocalPlayer:WaitForChild("PlayerGui")
	end
end

local function Corner(parent, radius)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, radius or 6)
	c.Parent = parent
	return c
end

local function Stroke(parent, color, thickness, transparency)
	local s = Instance.new("UIStroke")
	s.Color = color or Color3.fromRGB(48, 48, 54)
	s.Thickness = thickness or 1
	s.Transparency = transparency or 0
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.Parent = parent
	return s
end

local function Pad(parent, t, b, l, r)
	local p = Instance.new("UIPadding")
	p.PaddingTop = UDim.new(0, t or 8)
	p.PaddingBottom = UDim.new(0, b or 8)
	p.PaddingLeft = UDim.new(0, l or 8)
	p.PaddingRight = UDim.new(0, r or 8)
	p.Parent = parent
	return p
end

local function ResolveIcon(icon)
	if not icon or icon == "" then return "•" end
	local s = tostring(icon)
	local map = {
		info = "ⓘ",
		house = "⌂",
		shield = "⬡",
		eye = "◎",
		earth = "🌐",
		book = "📖",
		stairs = "🪜",
		settings = "⚙",
	}
	if map[s] then return map[s] end
	if s:match("^rbxassetid://") then return "❖" end
	return s
end

	local function beginDrag(input)
		if self._locked then return end
		dragging = true
		dragStart = input.Position
		startPos = Main.Position
	end

		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

		end

	Header.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			beginDrag(input)
		end
	end)

	ResizeBtn.InputBegan:Connect(function(input)
		if self._locked then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			resizing = true
			resizeStart = input.Position
			startSize = Main.Size
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			if dragging and dragStart and startPos then
				local delta = input.Position - dragStart
				Main.Position = UDim2.new(
					startPos.X.Scale,
					startPos.X.Offset + delta.X,
					startPos.Y.Scale,
					startPos.Y.Offset + delta.Y
				)
			elseif resizing and resizeStart and startSize then
				local delta = input.Position - resizeStart
				local w = math.clamp(startSize.X.Offset + delta.X, self.MinSize.X, self.MaxSize.X)
				local h = math.clamp(startSize.Y.Offset + delta.Y, self.MinSize.Y, self.MaxSize.Y)
				Main.Size = UDim2.new(0, w, 0, h)
			end
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
			resizing = false
		end
	end)

		end
	end)

			for _, t in ipairs(self.Tabs) do
				t.Page.Visible = false
				t.Button.BackgroundTransparency = 1
				t.Button:FindFirstChildWhichIsA("TextLabel").TextColor3 = Color3.fromRGB(150, 150, 160)
				for _, child in ipairs(t.Button:GetChildren()) do
					if child:IsA("TextLabel") then
						child.TextColor3 = Color3.fromRGB(150, 150, 160)
					end
				end
			end
			page.Visible = true
			btn.BackgroundTransparency = 0
			for _, child in ipairs(btn:GetChildren()) do
				if child:IsA("TextLabel") then
					child.TextColor3 = Color3.fromRGB(242, 242, 248)
				end
			end
			self.ActiveTab = tab
		end

			function gb:AddToggle(id, opts)
				opts = opts or {}
				local state = { Value = opts.Default and true or false, _cbs = {} }

				local row = Instance.new("TextButton")
				row.BackgroundColor3 = Color3.fromRGB(21, 21, 26)
				row.Size = UDim2.new(1, 0, 0, 28)
				row.Text = ""
				row.Parent = list
				Corner(row, 5)

				local tLbl = Instance.new("TextLabel")
				tLbl.BackgroundTransparency = 1
				tLbl.Position = UDim2.new(0, 8, 0, 0)
				tLbl.Size = UDim2.new(1, -48, 1, 0)
				tLbl.Font = Enum.Font.Gotham
				tLbl.Text = tostring(opts.Text or id)
				tLbl.TextColor3 = opts.Risky and Color3.fromRGB(255, 110, 110) or Color3.fromRGB(224, 224, 232)
				tLbl.TextSize = 12
				tLbl.TextXAlignment = Enum.TextXAlignment.Left
				tLbl.Parent = row

				local pill = Instance.new("Frame")
				pill.Position = UDim2.new(1, -34, 0.5, -8)
				pill.Size = UDim2.new(0, 26, 0, 16)
				pill.Parent = row
				Corner(pill, 8)

				local dot = Instance.new("Frame")
				dot.Size = UDim2.new(0, 12, 0, 12)
				dot.Position = UDim2.new(0, 2, 0.5, -6)
				dot.BackgroundColor3 = Color3.fromRGB(245, 245, 252)
				dot.Parent = pill
				Corner(dot, 6)

				local function refresh()
					if state.Value then
						pill.BackgroundColor3 = Color3.fromRGB(70, 200, 120)
						dot.Position = UDim2.new(1, -14, 0.5, -6)
					else
						pill.BackgroundColor3 = Color3.fromRGB(54, 54, 64)
						dot.Position = UDim2.new(0, 2, 0.5, -6)
					end
				end

				function state:SetValue(v)
					state.Value = v and true or false
					refresh()
					for _, cb in ipairs(state._cbs) do pcall(cb, state.Value) end
				end

				function state:OnChanged(cb)
					table.insert(state._cbs, cb)
					if state.Value then pcall(cb, state.Value) end
				end

				function state:AddKeyPicker() return state end
				function state:AddColorPicker() return state end

				refresh()
				row.MouseButton1Click:Connect(function() state:SetValue(not state.Value) end)

				self.Toggles[id] = state
				table.insert(self._searchable, { Box = box, Text = tostring(opts.Text or id) })
				return state
			end

			function gb:AddSlider(id, opts)
				opts = opts or {}
				local minV = tonumber(opts.Min) or 0
				local maxV = tonumber(opts.Max) or 100
				local defV = tonumber(opts.Default) or minV
				local state = { Value = math.clamp(defV, minV, maxV), _cbs = {} }

				local wrap = Instance.new("Frame")
				wrap.BackgroundColor3 = Color3.fromRGB(21, 21, 26)
				wrap.Size = UDim2.new(1, 0, 0, 38)
				wrap.Parent = list
				Corner(wrap, 5)

				local sLbl = Instance.new("TextLabel")
				sLbl.BackgroundTransparency = 1
				sLbl.Position = UDim2.new(0, 8, 0, 3)
				sLbl.Size = UDim2.new(1, -16, 0, 16)
				sLbl.Font = Enum.Font.Gotham
				sLbl.Text = string.format("%s: %s", tostring(opts.Text or id), tostring(state.Value))
				sLbl.TextColor3 = Color3.fromRGB(218, 218, 228)
				sLbl.TextSize = 11
				sLbl.TextXAlignment = Enum.TextXAlignment.Left
				sLbl.Parent = wrap

				local track = Instance.new("Frame")
				track.BackgroundColor3 = Color3.fromRGB(40, 40, 48)
				track.Position = UDim2.new(0, 8, 0, 24)
				track.Size = UDim2.new(1, -16, 0, 7)
				track.Parent = wrap
				Corner(track, 4)

				local fill = Instance.new("Frame")
				fill.BackgroundColor3 = Color3.fromRGB(115, 115, 225)
				fill.Size = UDim2.new(0, 0, 1, 0)
				fill.Parent = track
				Corner(fill, 4)

				local function setAlpha(a)
					a = math.clamp(a, 0, 1)
					local raw = minV + (maxV - minV) * a
					local val = opts.Rounding == 1 and (math.floor(raw * 10 + 0.5) / 10) or math.floor(raw + 0.5)
					state.Value = math.clamp(val, minV, maxV)
					local ratio = (maxV == minV) and 0 or ((state.Value - minV) / (maxV - minV))
					fill.Size = UDim2.new(math.clamp(ratio, 0, 1), 0, 1, 0)
					sLbl.Text = string.format("%s: %s", tostring(opts.Text or id), tostring(state.Value))
					for _, cb in ipairs(state._cbs) do pcall(cb, state.Value) end
				end

				setAlpha((maxV == minV) and 0 or ((state.Value - minV) / (maxV - minV)))

				local sliding = false
				track.InputBegan:Connect(function(input)
					if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
						sliding = true
						setAlpha((input.Position.X - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X))
					end
				end)
				UserInputService.InputChanged:Connect(function(input)
					if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
						setAlpha((input.Position.X - track.AbsolutePosition.X) / math.max(1, track.AbsoluteSize.X))
					end
				end)
				UserInputService.InputEnded:Connect(function(input)
					if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
						sliding = false
					end
				end)

				function state:OnChanged(cb) table.insert(state._cbs, cb) end
				self.Options[id] = state
				table.insert(self._searchable, { Box = box, Text = tostring(opts.Text or id) })
				return state
			end

			function gb:AddDropdown(id, opts)
				opts = opts or {}
				local vals = opts.Values or {}
				local def = (type(opts.Default) == "string" and opts.Default) or vals[tonumber(opts.Default) or 1] or ""
				local state = { Value = def, _cbs = {} }

				local btn = Instance.new("TextButton")
				btn.BackgroundColor3 = Color3.fromRGB(23, 23, 29)
				btn.Size = UDim2.new(1, 0, 0, 28)
				btn.Font = Enum.Font.GothamMedium
				btn.Text = string.format("%s: %s", tostring(opts.Text or id), tostring(state.Value))
				btn.TextColor3 = Color3.fromRGB(224, 224, 232)
				btn.TextSize = 12
				btn.Parent = list
				Corner(btn, 5)

				btn.MouseButton1Click:Connect(function()
					if #vals == 0 then return end
					local idx = 1
					for i, v in ipairs(vals) do
						if v == state.Value then idx = i break end
					end
					state.Value = vals[(idx % #vals) + 1]
					btn.Text = string.format("%s: %s", tostring(opts.Text or id), tostring(state.Value))
					for _, cb in ipairs(state._cbs) do pcall(cb, state.Value) end
				end)

				function state:OnChanged(cb) table.insert(state._cbs, cb) end
				function state:SetValues(nv)
					vals = nv or {}
					if #vals > 0 and not table.find(vals, state.Value) then
						state.Value = vals[1]
						btn.Text = string.format("%s: %s", tostring(opts.Text or id), tostring(state.Value))
					end
				end

				self.Options[id] = state
				table.insert(self._searchable, { Box = box, Text = tostring(opts.Text or id) })
				return state
			end

			function gb:AddInput(id, opts)
				opts = opts or {}
				local state = { Value = tostring(opts.Default or ""), _cbs = {} }

				local tb = Instance.new("TextBox")
				tb.BackgroundColor3 = Color3.fromRGB(23, 23, 29)
				tb.Size = UDim2.new(1, 0, 0, 28)
				tb.Font = Enum.Font.Gotham
				tb.PlaceholderText = tostring(opts.Placeholder or opts.Text or id)
				tb.PlaceholderColor3 = Color3.fromRGB(120, 120, 130)
				tb.Text = state.Value
				tb.TextColor3 = Color3.fromRGB(235, 235, 242)
				tb.TextSize = 12
				tb.ClearTextOnFocus = false
				tb.Parent = list
				Corner(tb, 5)
				Stroke(tb, Color3.fromRGB(48, 48, 58), 1, 0.2)

				tb:GetPropertyChangedSignal("Text"):Connect(function()
					state.Value = tb.Text
					for _, cb in ipairs(state._cbs) do pcall(cb, state.Value) end
				end)

				function state:OnChanged(cb) table.insert(state._cbs, cb) end
				self.Options[id] = state
				table.insert(self._searchable, { Box = box, Text = tostring(opts.Text or id) })
				return state
			end

			function gb:AddKeybind(id, opts)
				opts = opts or {}
				local state = { Value = opts.Default or Enum.KeyCode.RightShift, _cbs = {} }

				local btn = Instance.new("TextButton")
				btn.BackgroundColor3 = Color3.fromRGB(23, 23, 29)
				btn.Size = UDim2.new(1, 0, 0, 28)
				btn.Font = Enum.Font.GothamMedium
				btn.Text = string.format("%s: [%s]", tostring(opts.Text or id), tostring(state.Value.Name))
				btn.TextColor3 = Color3.fromRGB(224, 224, 232)
				btn.TextSize = 12
				btn.Parent = list
				Corner(btn, 5)

				local listening = false
				btn.MouseButton1Click:Connect(function()
					listening = true
					btn.Text = string.format("%s: [...]", tostring(opts.Text or id))
				end)

				UserInputService.InputBegan:Connect(function(input, gp)
					if gp then return end
					if listening and input.UserInputType == Enum.UserInputType.Keyboard then
						listening = false
						state.Value = input.KeyCode
						btn.Text = string.format("%s: [%s]", tostring(opts.Text or id), tostring(state.Value.Name))
					elseif not listening and input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == state.Value then
						for _, cb in ipairs(state._cbs) do pcall(cb) end
					end
				end)

				function state:OnClick(cb) table.insert(state._cbs, cb) end
				self.Options[id] = state
				table.insert(self._searchable, { Box = box, Text = tostring(opts.Text or id) })
				return state
			end

			return gb
		end

		function tab:AddRightTabbox(title) return makeGroupbox(rightCol, title, "◈") end

		return tab
	end
